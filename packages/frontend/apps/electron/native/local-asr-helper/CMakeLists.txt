cmake_minimum_required(VERSION 3.16)
project(nota_local_asr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${WHISPER_SOURCE} whisper)
add_executable(nota-whisper-helper main.cpp)
target_link_libraries(nota-whisper-helper PRIVATE whisper)
if(NEEDLE_LIBRARY)
  add_executable(nota-whistle-helper main.cpp)
  target_compile_definitions(nota-whistle-helper PRIVATE NOTA_WHISTLE)
  target_include_directories(nota-whistle-helper PRIVATE ${NEEDLE_INCLUDE})
  target_link_libraries(nota-whistle-helper PRIVATE ${NEEDLE_LIBRARY})
  if(APPLE)
    target_link_libraries(nota-whistle-helper PRIVATE "-framework Accelerate")
  elseif(UNIX)
    target_link_libraries(nota-whistle-helper PRIVATE pthread m)
  endif()
endif()
