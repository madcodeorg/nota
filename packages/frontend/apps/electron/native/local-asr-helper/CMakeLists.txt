cmake_minimum_required(VERSION 3.16)
project(nota_local_asr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
# Metal on Apple (shaders embedded, so the helper stays a single binary).
# Windows/Linux GPU (Vulkan) is opt-in at build time via NOTA_WHISPER_VULKAN.
if(APPLE)
  set(GGML_METAL ON CACHE BOOL "" FORCE)
  set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
else()
  set(GGML_METAL OFF CACHE BOOL "" FORCE)
endif()
option(NOTA_WHISPER_VULKAN "Build whisper.cpp with the Vulkan backend" OFF)
set(GGML_VULKAN ${NOTA_WHISPER_VULKAN} CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${WHISPER_SOURCE} whisper)
add_executable(nota-whisper-helper main.cpp)
target_link_libraries(nota-whisper-helper PRIVATE whisper)
